import 'dart:async';
import 'dart:convert';
import 'dart:typed_data';
import 'dart:ui' as ui;

import 'package:catalogo_legum_pro/contracts.dart';
import 'package:catalogo_legum_pro/ui/widgets.dart';
import 'package:flutter/material.dart';
import 'package:flutter/rendering.dart';
import 'package:flutter_test/flutter_test.dart';

class _SyntheticMedia implements MediaManager {
  _SyntheticMedia(this.bytes);
  final Uint8List bytes;
  @override
  ImageProvider imageProvider(String relativePath) => MemoryImage(bytes);
  @override
  dynamic noSuchMethod(Invocation invocation) => super.noSuchMethod(invocation);
}

void main() {
  testWidgets('baseline actual ProductPhoto decode at 352x264 with DPR 3', (
    tester,
  ) async {
    tester.view.devicePixelRatio = 3;
    tester.view.physicalSize = const Size(1200, 1800);
    addTearDown(tester.view.resetDevicePixelRatio);
    addTearDown(tester.view.resetPhysicalSize);
    final cache = PaintingBinding.instance.imageCache;
    cache.clear();
    cache.clearLiveImages();
    final bytes = await tester.runAsync(() async {
      final recorder = ui.PictureRecorder();
      final canvas = Canvas(recorder);
      canvas.drawRect(
        const Rect.fromLTWH(0, 0, 4000, 3000),
        Paint()..color = const Color(0xff237c45),
      );
      final picture = recorder.endRecording();
      final image = await picture.toImage(4000, 3000);
      final data = await image.toByteData(format: ui.ImageByteFormat.png);
      image.dispose();
      picture.dispose();
      return data!.buffer.asUint8List();
    });
    final media = _SyntheticMedia(bytes!);
    await tester.pumpWidget(
      MaterialApp(
        home: Scaffold(
          body: Center(
            child: SizedBox(
              width: 352,
              child: ProductPhoto(
                media: media,
                image: const ProductImage(
                  id: 'baseline',
                  path: 'synthetic.png',
                ),
                fit: BoxFit.cover,
              ),
            ),
          ),
        ),
      ),
    );
    final imageWidget = tester.widget<Image>(find.byType(Image));
    final decoded = await tester.runAsync(() async {
      final stream = imageWidget.image.resolve(
        createLocalImageConfiguration(tester.element(find.byType(Image))),
      );
      final completer = Completer<ImageInfo>();
      late final ImageStreamListener listener;
      listener = ImageStreamListener(
        (info, _) {
          if (!completer.isCompleted) completer.complete(info.clone());
        },
        onError: (Object error, StackTrace? stack) {
          if (!completer.isCompleted) completer.completeError(error, stack);
        },
      );
      stream.addListener(listener);
      try {
        return await completer.future;
      } finally {
        stream.removeListener(listener);
      }
    });
    await tester.pump();
    final rendered = tester.renderObject<RenderImage>(find.byType(RawImage));
    expect(rendered.size, const Size(352, 264));
    expect(rendered.image, isNotNull);
    expect(decoded!.image.width, 4000);
    expect(decoded.image.height, 3000);
    expect(rendered.image!.width, decoded.image.width);
    expect(rendered.image!.height, decoded.image.height);
    // This is a decode-resolution observation, not a native frame-time benchmark.
    debugPrint(
      jsonEncode({
        'scenario': 'host_original_decode_baseline',
        'logicalBounds': [rendered.size.width, rendered.size.height],
        'dpr': 3,
        'physicalBounds': [1056, 792],
        'decodedDimensions': [decoded.image.width, decoded.image.height],
        'decodedRgbaBytes': decoded.image.width * decoded.image.height * 4,
        'imageCacheBytes': cache.currentSizeBytes,
        'imageCacheEntries': cache.currentSize,
      }),
    );
    decoded.dispose();
    await tester.pumpWidget(const SizedBox.shrink());
    cache.clear();
    cache.clearLiveImages();
  });
}
